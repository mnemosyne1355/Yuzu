import fs from "fs";
const P = "./";
const N = "new/";
const rd = (f) => { const s = fs.readFileSync(P + f, "utf8"); if (s.includes("\r\n")) throw new Error("CRLF in " + f); return s; };
const nw = (f) => fs.readFileSync(N + f, "utf8");
const writes = [];
const once = (s, find, repl, label) => { const i = s.indexOf(find); if (i < 0) throw new Error("NOT FOUND: " + label); if (s.indexOf(find, i + 1) >= 0) throw new Error("NOT UNIQUE: " + label); return s.slice(0, i) + repl + s.slice(i + find.length); };
const all = (s, find, repl, n, label) => { const parts = s.split(find); if (parts.length - 1 !== n) throw new Error("EXPECTED " + n + " GOT " + (parts.length - 1) + ": " + label); return parts.join(repl); };
const span = (s, a, b, repl, label) => { const i = s.indexOf(a); if (i < 0) throw new Error("NOT FOUND a: " + label); const j = s.indexOf(b, i + a.length); if (j < 0) throw new Error("NOT FOUND b: " + label); return s.slice(0, i) + repl + s.slice(j); };
const spanIncl = (s, a, b, repl, label) => { const i = s.indexOf(a); if (i < 0) throw new Error("NOT FOUND a: " + label); const j = s.indexOf(b, i + a.length); if (j < 0) throw new Error("NOT FOUND b: " + label); return s.slice(0, i) + repl + s.slice(j + b.length); };
